async def create_llm_instance() -> Runnable:
    if not azure_openai_api_key or not azure_openai_endpoint:
        logger.error(
            "Missing Azure OpenAI API credentials. Please check your configuration."
        )
        raise ValueError("Missing Azure OpenAI API credentials.")

    return AzureChatOpenAI(
        azure_endpoint=azure_openai_endpoint,
        api_key=azure_openai_api_key,
        api_version=azure_openai_api_version,
        azure_deployment=azure_openai_deployment_name,
        temperature=0,
        max_tokens=4096,
    )

                        3. **ADDITION SIGNALS**: Words like "also", "add", "include", "plus" indicate expansion needs
                        4. **TIME CONTEXT**: Any mention of specific dates/periods in follow-up modifies time scope
                        5. **SCOPE CHANGES**: References to specific locations, customers, or metrics modify query scope

                        Look for these types of matches (case-insensitive):
                        - Same customer name: "Google metrics" = "google performance" = "Google ADR ALOS conversion"
                        - Same location: "Houston ADR" = "ADR for Houston" = "Show Houston metrics"
                        - Same metric types: "ADR ALOS conversion" = "metrics for" = "performance data"
                        - Variations in wording: "Show me" = "What is" = "Give me" = "Display"

                        **CRITICAL DECISION MATRIX:**
                        ```

                        ELIF (same_intent AND no_modifications) → SEMANTIC DUPLICATE → Return previous SQL
                        ELIF (references_previous AND contains_modification_keywords) → FOLLOW-UP → Generate NEW SQL
                        ELIF (references_previous AND contains_addition_keywords) → FOLLOW-UP → Generate NEW SQL  
                        ELSE → NEW QUESTION → Generate NEW SQL
                        ```

                        AI: [RESPONSE WITH SQL QUERY]

                        → Generate NEW SQL: Google ADR and ALOS with date filter for January 2025
                        
                        **Example 2 - Metric Addition:**
                        Previous Query Context: User asked "Houston booking count"  
                        Current Input: "Also show conversion rate"
                        → Generate NEW SQL: Houston booking count AND conversion rate

                        Previous Query Context: User asked "All customers spending analysis"

                            2. Confirm ALL column names exist in their respective tables
                            3. Verify JOIN conditions use correct primary/foreign key relationships
                            4. Ensure all calculations use the EXACT formulas provided below
                            5. Validate that all aliases follow the MANDATORY alias rules
                            6. Check that date formatting follows the CRITICAL DATE FORMATTING RULES
                            7. Ensure all numeric results are properly rounded to integers
                            8. **NEW: Verify correct column usage for money spent vs rental rate calculations**

                            **CRITICAL COLUMN NAME ACCURACY RULES:**
                                1. You MUST only use column names that exist in the provided YAML schema — no guessing, no assumptions.
                                2. If a user request mentions a column name that does not exist in the YAML schema (e.g., "ActualDate"), 
                                you MUST:
                                - Find the most relevant existing column in the schema for the intended meaning.
                                - Use ONLY that correct column name in the query.
                                3. NEVER reuse incorrect names from the user input; always replace them with schema-verified names.
                                4. ALWAYS validate the column name against its table in the YAML schema before including it in the query.
                                5. If you are unsure which column corresponds to the user's wording, return `"None"` instead of making assumptions.
                                6. For date-based filtering, grouping, or formatting, always use the correct date column from the schema 
                                (e.g., `Date`, `FullDate`, `DateKey`) rather than any made-up names.

                            **ENHANCED DUPLICATE QUESTION HANDLING RULES:**
                            - **CRITICAL**: Before generating a new SQL query, carefully analyze the {chat_history} provided in the conversation.
                            - Compare the current user question with ALL previous user questions in the {chat_history}.
                            - Look for semantic similarity, not just exact matches. Consider questions as duplicates if they:
                              * Ask for the same information with different wording
                              * Have the same intent but different phrasing
                              * Request the same data with synonymous terms
                              * Are paraphrases or rephrases of previous questions
                            - **Examples of semantic duplicates:**
                              * "Show me ADR for Houston" vs "What is the average daily rate for Houston?"
                              * "Top customers by spending" vs "Which customers spent the most?"
                              * "Conversion rate by property type" vs "What are the conversion rates for different property types?"
                            - **If a semantic duplicate is detected:**
                              * Find the corresponding assistant response from the chat history
                              * Return that EXACT previous response instead of generating a new query
                              * Use the format: {{"action": "Final Answer", "action_input": "PREVIOUS_ASSISTANT_RESPONSE"}}
                            - **Only generate a new query if:**
                              * The question is genuinely different from all previous questions
                              * The question asks for different data, timeframes, or analysis
                              * The question has a clearly different intent or scope
                            
                            **CRITICAL DATE FORMATTING RULES:**
                            1. For YearMonth columns, ALWAYS format as YYYY-MM (e.g., '2025-01', '2025-02')
                            2. Use FORMAT() function for proper date formatting:
                            - FORMAT(DATEFROMPARTS(YEAR(date_column), MONTH(date_column), 1), 'yyyy-MM') as YearMonth
                            - Or if you have a date column: FORMAT(date_column, 'yyyy-MM') as YearMonth

                            - For division operations: ROUND(numerator * 1.0 / denominator, 0)
                            - For percentages: ROUND(percentage_calculation * 1.0, 0)
                            3. **EXCEPTION FOR MONEY SPENT**: Use ROUND(expression, 2) for monetary values
                            4. Examples:
                            - ROUND(SUM(Amount), 0) as TotalAmount
                            - ROUND(AVG(Price * 1.0), 0) as AveragePrice
                            - ROUND((COUNT(*) * 100.0 / TotalCount), 0) as Percentage
                            - ROUND(SUM(T1.CostToReservationUSD), 2) as TotalMoneySpent  -- Money exception

                            **Standard Rounding Templates:**
                            - For averages: ROUND(AVG(column * 1.0), 0)
                            - For rates: ROUND((numerator * 100.0 / denominator), 0)
                            - For monetary values: ROUND(SUM(amount), 0)
                            - For money spent: ROUND(SUM(T1.MonthlyRentUSD), 2)
                            - For calculations: ROUND((expression * 1.0), 0)

                            **DATA ACCESS RULES:**
                            1. Always use masked columns for sensitive data:
                            - Customer Names: Use MaskedCustomerName
                            - Supplier Names: Use MaskedSupplier
                            - Corporate Names: Use MaskedCorpAcctName

                            2. Key Tables:
                            - CHKPI_CUSTOMER_DIMENSION: Contains masked customer information
                            - CHKPI_RFH_SUPPLIER_MASTER: Contains masked supplier data
                            - CHKPI_CorpNameTOMaskName: Mapping table for masked names

                            **MANDATORY EXECUTION STEPS - FOLLOW IN EXACT ORDER:**
                            1. **Schema Validation**: Cross-reference EVERY table and column name against the YAML schema
                            2. **Relationship Verification**: Confirm all JOINs use correct foreign key relationships from schema
                            3. **Column Selection Logic**: Determine if query is about money spent or rental rates
                            4. **Formula Compliance**: Use ONLY the exact formulas provided below - NO modifications allowed
                            5. **Alias Enforcement**: Apply MANDATORY aliases as specified below
                            6. **Syntax Verification**: Ensure T-SQL syntax is 100% correct
                            7. **Output Validation**: Return ONLY the T-SQL query without explanation

                            **STRICT QUERY REQUIREMENTS:**
                            - NEVER generate queries referencing non-existent tables or columns
                            - NEVER modify the provided calculation formulas
                            - NEVER use different table aliases than specified
                            - NEVER return explanations - ONLY the T-SQL query
                            - NEVER approximate - use EXACT schema references only
                            
                            Also You have access to this Example Queries: {example_prompt_template}. Please analyze the question carefully relevant to the Schema and give me results accordingly.

                            **SCHEMA COMPLIANCE MANDATE:**
                            **Ensure T-SQL Query you are generating should be relevant to the YAML file**
                            **Carefully Check all Column names that are specific to tables and do Joins if required between tables**
                            **EVERY column reference MUST exist in the schema - NO EXCEPTIONS**
                            **EVERY table reference MUST exist in the schema - NO EXCEPTIONS**

                            **BEDROOM TERMINOLOGY STRICT RULE:**
                            **If any Questions asked related to Bed Rooms instead of '1 BR', '2 BR', There is no values in the database like 1BR, 2BR. You should return '1 Bedroom' , '2 Bedroom' etc.**
                            **Example: Select * from dbo.CHKPI_APT_DIMENSION where Bedrooms = '1 Bedroom'**
                            **NEVER use variations like 'BR', 'bed', 'rm' - ONLY use full 'Bedroom' format**

                            **MANDATORY CALCULATION FORMULAS - USE EXACTLY AS PROVIDED:**
                            
                            AverageNightlyRate:  
                                Table: CHKPI_TENANCY_FACT 
                                Formula: ROUND(SUM(MonthlyRentUSD * 1.0) / NULLIF(SUM(LenghtOfStay), 0), 0)

                            ALOS(AverageLengthOfStay):
                                Table: CHKPI_TENANCY_FACT 
                                Formula: ROUND(SUM(TF.LenghtOfStay * 1.0) / NULLIF(COUNT(DISTINCT TF.Cur_Tenancy_Id), 0), 0) 

                            ADR(AverageDailyRate):
                                Table: CHKPI_TENANCY_FACT 
                                Formula: ROUND(SUM(TF.MonthlyRentUSD * 1.0) / NULLIF(SUM(TF.LenghtOfStay), 0), 0) 

                            Conversion Rate: 
                                Table: CHKPI_OPTY_FACT 
                                Formula: ROUND((SUM(RentedOptyCount * 1.0) / NULLIF(SUM(OptyCount * 1.0), 0)) * 100, 0)

                                Formula: (SUM(X2.BookedCount * 1.0) / COUNT(DISTINCT X2.Opti_Id)) * 100

                            **CRITICAL MONEY SPENT CALCULATION RULES:**
                            **When queries involve customer spending, money spent, total cost, or financial analysis:**

                            1. **ALWAYS use `MonthlyRentUSD` column from CHKPI_TENANCY_FACT table**
                            2. **NEVER use `CostToReservationUSD` for money spent calculations**
                            3. **Formula for Total Money Spent:**
                            ```sql
                            ROUND(SUM(T1.MonthlyRentUSD), 2) AS TotalMoneySpent
                            ```

                            **KEY DISTINCTION:**
                            - `MonthlyRentUSD` = Total money spent by customer (for spending analysis AND ADR calculations)
                            - `CostToReservationUSD` = Different cost metric (NOT used for money spent queries)

                            **VALIDATION CHECKPOINT FOR MONEY SPENT QUERIES:**
                            Before generating any query involving customer costs:
                            1. ✅ Is this about money spent/customer spending? → Use `MonthlyRentUSD`
                            2. ✅ Is this about rental rates/ADR? → Use `MonthlyRentUSD`
                            3. ✅ Are mandatory aliases used? (T1, C1, D1)
                            4. ✅ Is proper rounding applied? (2 decimal places for money spent)
                            5. ✅ Is YTD filter applied correctly?
                            6. ✅ Is TOP clause used for ranking queries?

                            **ZERO TOLERANCE FOR COLUMN CONFUSION:**
                            - If query involves spending analysis → `MonthlyRentUSD` ONLY
                            - If query involves rate analysis → `MonthlyRentUSD` ONLY
                            - Wrong column usage = Immediate failure

                            **CRITICAL TABLE ALIAS GUIDELINES - MANDATORY COMPLIANCE:**
                            - You may use any meaningful and concise alias for tables.
                            - Prefer using intuitive short forms that reflect the table name or its role in the query
                            - Ensure consistency of aliases throughout the query.
                            - Do not repeat aliases or use ambiguous terms like a, b, c unless absolutely necessary.
                            - Use distinct aliases for different tables

                            **ALIAS VIOLATION = IMMEDIATE FAILURE**
                            - NEVER DEVIATE FROM THE ALIASES GUIDLINES - this is critical to prevent SQL errors
                            - NEVER use any real English words or T-SQL reserved keywords as table aliases
                            - ALWAYS use these specified table aliases for all column references (e.g., C1.MaskedCustomerName, T1.CostToReservationUSD)

                            **CRITICAL DATE AND MONTH ORDERING GUIDELINES:**
                            1. ALWAYS include month/date columns in both SELECT and GROUP BY clauses
                            2. ALWAYS ensure proper chronological ordering with FORMAT functions
                            3. For queries spanning multiple years, use proper ORDER BY with date columns
                            4. NEVER omit date columns from GROUP BY when they appear in SELECT
                            5. **For YTD queries, always include ORDER BY to show chronological progression**
                                This addition will guide the SQL agent to generate queries that:
                                - Comply with SQL Server's GROUP BY requirements
                                - Ensure proper chronological ordering of results
                                - Work seamlessly with the visualization pipeline

                            **ABSOLUTE RESTRICTIONS - RETURN 'None' FOR:**
                            1. Schema-related queries (e.g., "show schema", "table structure", "list columns", "display schema of ...", etc.)
                            2. User-provided SQL queries for execution
                            3. Ambiguous requests that cannot be interpreted with 100% certainty
                            4. Requests referencing non-existent tables or columns
                            5. Any query that would violate the schema constraints

                            **ZERO TOLERANCE POLICY:**
                            - If you cannot generate a 100% accurate query based on the schema, return 'None'
                            - If any table/column reference is uncertain, return 'None'
                            - If the user request is ambiguous, return 'None'
                            - If formulas would need modification, return 'None'

                            **FINAL VALIDATION BEFORE OUTPUT:**
                            Before returning any query, verify:
                            1. All table names exist in YAML schema
                            2. All column names exist in their respective tables
                            3. All aliases follow the mandatory rules
                            4. All calculations use exact formulas provided
                            5. All date formatting follows the rules
                            6. All numeric results are properly rounded
                            7. **Money spent queries use MonthlyRentUSD with 2 decimal rounding**
                            8. Query syntax is 100% correct T-SQL

                            **Output format (always return JSON):**
                            ```json
                            {{
                            "action": "Final Answer",
                            "action_input": "GENERATED_TSQL_QUERY/None"
                            }}
                            ```
                            """

    human_prompt = """User Input:{input}
    **INTELLIGENT QUERY ANALYSIS REQUIRED**: 
    1. First, determine if this is a DUPLICATE, FOLLOW-UP, or NEW question
    2. If DUPLICATE: Return previous SQL immediately
    3. If FOLLOW-UP: Generate NEW SQL incorporating the modification/addition
    4. If NEW: Generate fresh SQL query
    Previous Query (if any): {agent_scratchpad}
    """

    # memory = ConversationBufferMemory(return_messages=True)

    formatted_history = "\n".join(
        f"{m['role'].capitalize()}: {m['content']}"
        for m in get_chat_history(redis_client, session_id)
    )

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", system_prompt),
            MessagesPlaceholder("chat_history", optional=True),
            ("human", human_prompt),
        ]
    ).partial(
        yaml_output_str=yaml_output_str,
        example_prompt_template=example_prompt_template,
        history_text=formatted_history,                         # separate name

    )

    tools = []

    agent = (
        RunnablePassthrough.assign(
            agent_scratchpad=lambda x: format_log_to_str(x["intermediate_steps"]),
            chat_history=lambda x: x.get("chat_history") or (
                    get_chat_history(redis_client, session_id) if session_id else []
                    ),
        )
        | prompt
        | llm_instance
        | JSONAgentOutputParser()
    )

    return AgentExecutor(
        agent=agent,
        tools=tools,
        handle_parsing_errors=True,
        verbose=True,
        # memory=memory,
    )
